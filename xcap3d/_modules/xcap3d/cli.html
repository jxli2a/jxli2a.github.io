"""The xcap3d command line: run, download, process, invert, replay, report, compare, catalog and build.

The output directory is `--out`, else `XCAP3D_OUT`, else `./xcap3d-out`. The reference catalog files are read from
the directory `--catalogs`, else `XCAP3D_CATALOGS`; without either, the record lists every reference as not given. The raw download goes
to `<out>/<id>/raw/` and the waveforms with the full instrument response removed to `<out>/<id>/data-full/`. With
`--label`, result.json, report.html and the figures go to `<out>/<id>/models/<label>/`, so several Green's-function
sets can be compared on one download; without it they go to `<out>/<id>/`.

Defaults: the station jackknife on.
"""

def out_dir(out=None):
    """The output directory: `out` (--out), else XCAP3D_OUT, else ./xcap3d-out; the variable is read at each call."""
    return Path(out or os.environ.get("XCAP3D_OUT") or "xcap3d-out")


def catalogs_dir(catalogs=None):
    """The directory of the reference catalog files (record.ref_files names them): `catalogs` (--catalogs), else
    XCAP3D_CATALOGS, else None; the variable is read at each call."""
    c = catalogs or os.environ.get("XCAP3D_CATALOGS")
    return Path(c) if c else None


DATA = "data-full"  # <out>/<id>/<this>: the waveforms an inversion reads, made from the raw download <id>/raw


def has_store(data):
    """True if `data` holds processed waveforms; SystemExit if it holds files but no meta.json."""
    data = Path(data)
    if (data / "meta.json").exists():
        return True
    if data.is_dir() and any(data.iterdir()):
        raise SystemExit(f"{data}: no meta.json: move it away and run xcap3d process again")
    return False


def read_store(event_id, data):
    """The meta of the processed waveforms in `data`; SystemExit if an xcap3d of another store version made them."""
    m = read_meta(data)
    if m.get("store_version", 1) != STORE_VERSION:
        raise SystemExit(
            f"{event_id}: {data} is a store of version {m.get('store_version', 1)}, this xcap3d reads version "
            f"{STORE_VERSION}: move it away and run xcap3d process {event_id} again"
        )
    return m


def write_json(path, rec):
    """Write to a temporary file and rename it, so that `path` always holds a whole record."""
    tmp = path.with_name(path.name + ".tmp")
    try:
        tmp.write_text(json.dumps(rec, indent=1))
        os.replace(tmp, path)
    finally:
        tmp.unlink(missing_ok=True)


def write_failed(dest, e, status, event_id, mode, environment, database, inputs, run_spec):
    """Write the record of a run that ended without a solution ('no_solution') or crashed ('error') and return it.

    The page and figures of an earlier run are removed with its record, so that nothing older stands for this run.
    """
    dest.mkdir(parents=True, exist_ok=True)
    report.clear_report(dest)  # the old page and figures would pair with this record
    failed = dict(
        event_id=event_id,
        mode=mode,
        status=status,
        error=type(e).__name__,
        reason=str(e),
        environment=environment,
        database=database,
        inputs=inputs,
        run_spec=run_spec,
    )
    write_json(dest / "result.json", failed)
    return failed


def environment(device):
    """Versions, device and threads of a run, and one sha256 over every .py file of the package.

    The source digest stands in for git describe, which an installed build does not have.
    """
    source = hashlib.sha256(
        b"".join(p.read_bytes() for p in sorted(Path(__file__).parent.rglob("*.py")))
    )
    versions = {}
    for name in ("xcap3d", "numpy", "scipy", "obspy", "h5py", "torch"):
        try:
            versions[name] = metadata.version(name)
        except metadata.PackageNotFoundError:
            versions[name] = None
    threads = ("OMP_NUM_THREADS", "MKL_NUM_THREADS", "OPENBLAS_NUM_THREADS")
    return dict(
        python=platform.python_version(),
        packages=versions,
        device=device or "cpu",
        threads={k: os.environ.get(k) for k in threads},
        source_sha256=source.hexdigest(),
    )


def input_digests(root):
    """The sha256 of every file of a download, and one digest of the whole set."""
    root = Path(root)
    files = {str(p.relative_to(root)): sha256(p) for p in sorted(root.rglob("*")) if p.is_file()}
    whole = "".join(f"{digest} {name}\n" for name, digest in files.items())
    return dict(root=str(root), digest=hashlib.sha256(whole.encode()).hexdigest(), files=files)

def run_event(
    event_id,
    gf,
    *,
    jackknife=None,
    edge_margin_km=10.0,
    near_field_cells=1,
    write_report=False,
    label=None,
    out=None,
    device="auto",
    release=None,
    catalogs=None,
    mt="dc",
    refine=False,
    grid_step=5.0,
    refine_step=REFINE_STEP_DEG,
    verbose=True,
):
    """Run `xcap3d run` from a script: download if needed, invert, write result.json and return the record.

    A production run reads `<out>/<id>/data-full`, made from the raw download `<out>/<id>/raw` when missing. A
    production run that ends without a solution (no station or depth left,
    a store that differs from the release, an incomplete download) writes a record with status 'no_solution' and the
    reason. A run that crashes writes a record with status 'error', the exception's type and message, and raises it.
    The record's run_spec names `mt`, `refine`, `grid_step` and `refine_step` only when they are not the
    defaults.

    Args:
        event_id: SCEDC event id.
        gf: the database, a directory of HDF5 stores.
        jackknife: None or True for the station jackknife (uncertainty.fast_jackknife), False for none.
        edge_margin_km: drop a station whose box edge is closer than this to the epicenter [km].
        near_field_cells: drop a station-centered box's station when the epicenter is within this many lattice
            cells of the station node.
        write_report: also write report.html next to result.json.
        label: name of the Green's-function set; the products go to `<out>/<id>/models/<label>/`.
        out: output directory; default `XCAP3D_OUT`, else ./xcap3d-out.
        device: "auto" (a GPU if torch sees one), "cpu" (or None) or "cuda:N", chosen once for every search of the
            run.
        release: the database's release file; default `<gf>/release.json`. A production run needs one.
        catalogs: directory of the reference catalog files; default `XCAP3D_CATALOGS`.
        mt: 'dc' (default), 'deviatoric' ('dev') or 'full'. The last two add the non-DC search at the chosen depth
            around the DC solution (orient.search_nondc) as the record's `non_dc` block; the DC solution is unchanged.
        refine: after each double-couple search, search 1-deg nodes within Kagan 8 deg of up to 3 coarse minima
            (orient.refine_search) and keep the lowest misfit.
        grid_step: spacing of the strike/dip/rake grid [deg], more than 0 and at most 30.
        refine_step: spacing of the fine nodes of `refine` [deg], less than half its radius (1.6 `grid_step`); only
            with `refine`.
        verbose: print each stage as it starts and ends (download, processing, Green's functions, QC, depth scan,
            moment tensor, jackknife); False prints only the solution line.

    Returns:
        The result record as written to result.json, a `Result`.

    Raises:
        SystemExit: an option has an invalid value, or the inversion cannot start.
    """
    gf, device, mt = Path(gf), device or "cpu", MT_SHORT.get(mt, mt)
    if mt not in MT_MODES:
        raise SystemExit(f"--mt {mt!r}: one of {', '.join(MT_MODES)}")
    if not (refine is True or refine is False):
        raise SystemExit(f"refine {refine!r}: True or False")
    if not 0.0 < grid_step <= 30.0:
        raise SystemExit(f"--grid-step {grid_step:g}: more than 0 and at most 30 deg")
    if refine_step != REFINE_STEP_DEG and not refine:
        raise SystemExit(
            "--refine-step sets the spacing of --refine: add --refine or drop --refine-step"
        )
    radius = refine_radius(grid_step)
    if refine and not 0.0 < refine_step < radius / 2:
        raise SystemExit(
            f"--refine-step {refine_step:g}: more than 0 and less than half the refine radius, {radius:g} deg"
        )
    catalogs = catalogs_dir(catalogs)
    if label and not re.fullmatch(report.LABEL_RE, label):
        raise SystemExit(f"--label {label!r}: letters, digits and . _ + - only")
    if device not in ("auto", "cpu") and not device.startswith("cuda:"):
        raise SystemExit(f"--device {device!r}: auto, cpu or cuda:N")
    if device.startswith("cuda:") and kernel.torch is None:
        raise SystemExit(f"--device {device}: needs PyTorch (pip install 'xcap3d[gpu]')")
    if not (jackknife is None or jackknife is True or jackknife is False):
        raise SystemExit(f"jackknife {jackknife!r}: None, True or False")
    used = pick_device() if device == "auto" else None if device == "cpu" else device
    env = environment(used)
    spec = dict(
        event_id=event_id,
        gf=str(gf.resolve()),
        bands="production",
        shear="corrected",
        qc="huber",
        jackknife=jackknife,
        edge_margin_km=edge_margin_km,
        near_field_cells=near_field_cells,
        label=label,
        device=device,
        out=str(out_dir(out).resolve()),
        processing="full",
    )
    if mt != "dc":  # a record without it ran double couples only
        spec["mt"] = mt
    if refine:  # a record without it searched the coarse nodes only
        spec["refine"] = True
    if grid_step != 5.0:  # a record without it searched the 5-deg grid
        spec["grid_step"] = grid_step
    if refine_step != REFINE_STEP_DEG:  # a record without it refined on the 1-deg lattice
        spec["refine_step"] = refine_step
    home = out_dir(out) / event_id
    home.mkdir(parents=True, exist_ok=True)
    dest = home / "models" / label if label else home  # the record, page and figures of this run
    try:
        rel = read_release(gf, release)
    except ReleaseError as e:
        raise SystemExit(f"{event_id}: {e}")
    data, raw = home / DATA, home / "raw"
    inputs, info = None, {}
    try:
        if not has_store(data):
            if not raw.exists():
                download(event_id, raw, verbose)
            process(raw, data, verbose)
        m = read_store(event_id, data)
        store = {"root": str(data), "meta": m["meta"]}
        inputs = dict(input_digests(data), processing=m["processing"])
        inputs["references"] = {
            k: sha256(p) for k, p in record.ref_files(catalogs).items() if p.exists()
        }
        spec["release"] = rel["file"]
        jk = jackknife is not False
        spec.update(jackknife=jk)  # as run, not "default"
        # a run killed before it writes (out of memory, a signal) must not leave the previous record standing
        (dest / "result.json").unlink(missing_ok=True)
        report.clear_report(dest)  # the old page and figures would pair with the new numbers
        sol, depth, sols, depths = run_from_download(
            gf,
            store,
            EventConfig(m["evlo"], m["evla"], m["evdp"], m["mag"]),
            qc=QCParams(),
            info=info,
            edge_margin_km=edge_margin_km,
            near_field_cells=near_field_cells,
            jackknife=jk,
            device=used,
            release=rel,
            mt=mt,
            refine=refine,
            grid_step=grid_step,
            refine_step=refine_step,
            verbose=verbose,
        )
    except (NoSolution, ReleaseError, IncompleteDownload) as e:
        print(f"{event_id}: {e}", file=sys.stderr)
        failed = write_failed(
            dest, e, "no_solution", event_id, "production", env, info.get("database"), inputs, spec
        )
        return Result(failed)
    except Exception as e:  # a crash: the record says so, then the exception goes on
        write_failed(
            dest, e, "error", event_id, "production", env, info.get("database"), inputs, spec
        )
        raise
    dest.mkdir(parents=True, exist_ok=True)  # <id>/models/<label> for a labeled run
    extra = dict(environment=env, database=info["database"], inputs=inputs, run_spec=spec)
    if label:
        extra = dict(greens=record.greens_info(gf, label), **extra)
    try:
        rec = record.production_record(
            sol,
            depth,
            sols,
            depths,
            m,
            event_id,
            data,
            catalogs,
            store_xazim(gf),
            qc_info=info,
            coords=station_coordinates(raw),
        )
        nd = {"non_dc": info["non_dc"]} if "non_dc" in info else {}
        result = dict(rec, **nd, **extra, status="complete")
        write_fits(
            dest / "fits.h5", info["fits"], info["run"]["dt"]
        )  # before the record, which names its digest
        result["products"] = {"fits.h5": fits_digest(info["fits"])}
    except Exception as e:  # keep the flat keys, say what failed
        flat = record.flat_result(event_id, "production", depth, sol)
        result = dict(
            flat, **extra, status="product_error", report_error=f"{type(e).__name__}: {e}"
        )
    write_json(dest / "result.json", result)
    if write_report:
        print(
            report.write_event_report(result, dest)
            if result.get("schema") == record.SCHEMA
            else f"{event_id}: no report ({result['report_error']})"
        )
    print(solution_line(record.flat_result(event_id, "production", depth, sol)))
    return Result(result)

def replay(result, out, data=None, device=None, catalogs=None):
    """Run a recorded production run again, offline, and return how the new record differs from the recorded one.

    The inputs and release file must match the recorded sha256. The records are compared without paths, clock, file
    times and environment (printed as notes); the fits are compared by digest. The settings come from this code, so a
    default that changed since the run shows as a difference.

    Args:
        result: path of the recorded result.json; the record must be complete.
        out: output directory for the replay.
        data: the download, if it has moved since the run; default the recorded path.
        device: default the device the run used.
        catalogs: directory of the reference catalog files; default `XCAP3D_CATALOGS`.

    Returns:
        The differences, empty when the records agree.
    """
    rec = json.loads(Path(result).read_text())
    if rec.get("status") != "complete":
        raise SystemExit(
            f"{result}: status {rec.get('status')!r}; only a complete record can be replayed"
        )
    spec, inputs = rec["run_spec"], rec["inputs"]
    root = Path(data or inputs["root"])
    want = rec["database"]["file_sha256"]
    if read_release(spec["gf"], spec["release"])["file_sha256"] != want:
        old = archived_release(spec["gf"], want)  # the database has a newer release since the run
        if old is None:
            raise SystemExit(
                f"{spec['release']}: the release file differs from the one {result} used"
            )
        spec = dict(spec, release=str(old))
    here = Path(out) / spec["event_id"]
    if (here / "models" / spec["label"] if spec.get("label") else here).resolve() == Path(
        result
    ).resolve().parent:
        raise SystemExit(
            f"--out {out}: the run's own output directory; the replay would replace the record it compares"
        )
    here.mkdir(parents=True, exist_ok=True)
    data = here / DATA
    if not data.exists():
        data.symlink_to(root.resolve())
    if input_digests(data)["digest"] != inputs["digest"]:  # the files the replay will read
        raise SystemExit(f"{data}: the input files differ from those of {result}")
    catalogs = catalogs_dir(catalogs)
    now = {k: sha256(p) for k, p in record.ref_files(catalogs).items() if p.exists()}
    if now != inputs.get("references", {}):
        print(
            f"note: the reference catalogs under {catalogs} differ from the run's; references may differ"
        )
    new = run_event(
        spec["event_id"],
        spec["gf"],
        out=out,
        device=device or rec["environment"]["device"],
        catalogs=catalogs,
        **{
            k: spec[k] for k in REPLAY_KEYS if k in spec
        },  # a key the record lacks: the default it ran with
    )
    new = json.loads(json.dumps(new))  # as written to result.json: numpy floats become floats
    for k in sorted(set(rec["environment"]) | set(new["environment"])):
        if rec["environment"].get(k) != new["environment"].get(k):
            print(
                f"note: environment {k}: {rec['environment'].get(k)} -> {new['environment'].get(k)}"
            )

    def comparable(r):  # what the run computed: no paths, clock, file times or environment
        r = without_volatile(r)
        r.pop("environment")
        times = ("data_files_written", "download_time")
        out = dict(
            r,
            provenance={k: v for k, v in r.get("provenance", {}).items() if k not in times},
            # no paths or device
            run_spec={k: v for k, v in r["run_spec"].items() if k not in SPEC_SKIP},
            inputs={k: v for k, v in r["inputs"].items() if k not in ("root", "references")},
        )
        if (
            "greens" in r
        ):  # paths as given: the replay is handed the recorded, resolved GF directory
            out["greens"] = {k: v for k, v in r["greens"].items() if k not in ("dir", "store")}
        return out

    return differences(comparable(rec), comparable(new))




def download(event_id, raw, verbose=True):
    """`fetch.download_raw` into `raw`, saying when it starts and what it got when it ends."""
    if verbose:
        print(f"{event_id}: download from SCEDC (CI, within 100 km) ...", flush=True)
    t0 = time.monotonic()
    _, inv, _ = download_raw(event_id, raw)
    if verbose:
        n = sum(len(net.stations) for net in inv)
        mb = sum(f.stat().st_size for f in Path(raw).iterdir()) / 1e6
        print(f"  {n} stations, {mb:.0f} MB ({elapsed(t0)})", flush=True)


def process(raw, data, verbose=True):
    """`fetch.process_raw` from `raw` into `data`, saying when it starts and what it kept when it ends."""
    if verbose:
        print("  processing: instrument response, rotation to Z/R/T ...", flush=True)
    t0 = time.monotonic()
    res = process_raw(raw, data)
    if verbose:
        kept, lost = len(res["meta"]), len(res["dropped"])
        print(f"  {kept} stations kept, {lost} dropped ({elapsed(t0)})", flush=True)
    return res

def download_event(event_id, out=None):
    """Download an event (QuakeML, StationXML and raw seismograms) into `<out>/<id>/raw`.

    Does nothing if that folder exists (`fetch.download_raw`).

    Args:
        event_id: SCEDC event id.
        out: output directory; default `XCAP3D_OUT`, else ./xcap3d-out.

    Returns:
        The raw folder.
    """
    raw = out_dir(out) / event_id / "raw"
    if raw.exists():
        print(f"{raw}: already downloaded")
        return raw
    download(event_id, raw)
    print(raw)
    return raw

def process_event(event_id, out=None):
    """Remove the full instrument response from `<out>/<id>/raw` into `<out>/<id>/data-full`, the folder `run` reads.

    Does nothing if that folder exists (`fetch.process_raw`), and stops if an xcap3d of another store version made it.

    Args:
        event_id: SCEDC event id.
        out: output directory; default `XCAP3D_OUT`, else ./xcap3d-out.

    Returns:
        The data-full folder.
    """
    home = out_dir(out) / event_id
    data = home / DATA
    if has_store(data):
        read_store(event_id, data)
        print(f"{data}: already processed")
        return data
    if not (home / "raw").exists():
        raise SystemExit(
            f"{event_id}: no download in {home / 'raw'}; run xcap3d download {event_id} first"
        )
    try:
        process(home / "raw", data)
    except FileExistsError as e:
        raise SystemExit(str(e))
    print(data)
    return data

def invert_event(event_id, gf, **settings):
    """Run `run_event` on waveforms already processed, never downloading.

    Args:
        event_id: SCEDC event id.
        gf: as in `run_event`.
        **settings: the other keyword arguments of `run_event`.

    Returns:
        The result record.

    Raises:
        SystemExit: `<out>/<id>/data-full` holds no processed waveforms.
    """
    home = out_dir(settings.get("out")) / event_id
    if not has_store(home / DATA):
        raise SystemExit(
            f"{event_id}: no processed waveforms in {home / DATA}; run xcap3d download and xcap3d "
            "process first, or xcap3d run"
        )
    return run_event(event_id, gf, **settings)

def run_options(r):
    """The arguments of `xcap3d run`, which `xcap3d invert` shares."""
    r.add_argument("event_id")
    r.add_argument(
        "--gf",
        required=True,
        type=Path,
        help="the database, a directory of HDF5 stores",
    )
    r.add_argument(
        "--mt",
        choices=("dc", "dev", "deviatoric", "full"),
        metavar="{dc,dev,full}",
        default="dc",
        help="dc: double couple only (default); dev (deviatoric): ISO fixed at 0, CLVD free; full: ISO and CLVD free. "
        "deviatoric and full also give the non-DC moment tensor at the chosen depth, around the DC solution, "
        "in the record's non_dc block",
    )
    r.add_argument(
        "--refine",
        action="store_true",
        help="after each double-couple search, also search 1-deg nodes within Kagan 8 deg of up to 3 coarse minima "
        "(within 2%% of the lowest misfit) and keep the lowest misfit",
    )
    r.add_argument(
        "--grid-step",
        type=float,
        default=5.0,
        metavar="DEG",
        help="spacing of the strike/dip/rake grid, more than 0 and at most 30 deg (default 5). "
        f"Cost: {COST}",
    )
    r.add_argument(
        "--refine-step",
        type=float,
        default=REFINE_STEP_DEG,
        metavar="DEG",
        help="spacing of the --refine nodes (default 1), less than half the refine radius, which is 1.6 times the grid "
        "step (8 deg at 5); only with --refine",
    )
    r.add_argument(
        "--jackknife",
        action=argparse.BooleanOptionalAction,
        help="default on: leave each QC input station out, re-run the QC, the inversion at "
        "the chosen depth and the depth choice; median/p90 Kagan and the depth SE go to result.json "
        "'uncertainty'; every grid search from per-component arrays computed once per depth",
    )
    r.add_argument(
        "--edge-margin-km",
        type=float,
        default=10.0,
        help="exclude a station whose box edge is closer than this to the epicenter (default 10)",
    )
    r.add_argument(
        "--near-field-cells",
        type=int,
        default=1,
        help="exclude a station-centered box's station when the epicenter is within this many "
        "lattice cells (2 km) of the station node (default 1)",
    )
    r.add_argument(
        "--report", action="store_true", help="also write report.html next to result.json"
    )
    r.add_argument(
        "--label", help="name of this Green's-function set: write under <id>/models/<label>/"
    )
    r.add_argument("--out", type=Path, metavar="DIR", help=OUT_HELP)
    r.add_argument(
        "--release", type=Path, help="the database's release file (default <gf>/release.json)"
    )
    r.add_argument(
        "--device",
        default="auto",
        help="auto (a GPU if torch sees one), cpu, or cuda:N, for every search of the run",
    )
    r.add_argument(
        "--catalogs",
        type=Path,
        metavar="DIR",
        help="directory holding the reference catalog files the record compares with (default XCAP3D_CATALOGS; "
        "docs/usage.md names the files)",
    )


def main(argv=None):
    argv = sys.argv[1:] if argv is None else list(argv)
    p = argparse.ArgumentParser(prog="xcap3d")
    sub = p.add_subparsers(dest="cmd", required=True)
    r = sub.add_parser("run", help="moment tensor and depth for one SCEDC event")
    run_options(r)
    dl = sub.add_parser(
        "download",
        help="the event, StationXML and raw seismograms from SCEDC into <out>/<event_id>/raw",
    )
    dl.add_argument("event_id")
    dl.add_argument("--out", type=Path, metavar="DIR", help=OUT_HELP)
    pr = sub.add_parser(
        "process",
        help="the raw download -> the waveforms an inversion reads, the full instrument response removed, in "
        "<out>/<event_id>/data-full",
    )
    pr.add_argument("event_id")
    pr.add_argument("--out", type=Path, metavar="DIR", help=OUT_HELP)
    iv = sub.add_parser(
        "invert",
        help="the inversion of run on waveforms already processed; the options of run, never a download",
    )
    run_options(iv)
    rp = sub.add_parser("report", help="report.html from an existing run's result.json")
    rp.add_argument("event_id")
    rp.add_argument(
        "--out",
        type=Path,
        metavar="DIR",
        help="the output directory holding <event_id>/result.json (default XCAP3D_OUT, else ./xcap3d-out)",
    )
    rp.add_argument("--label", help="the model run under <id>/models/<label>/")
    cp = sub.add_parser(
        "compare", help="compare.html: every <id>/models/<label>/ of one event in one page"
    )
    cp.add_argument("event_id")
    cp.add_argument("--out", type=Path, metavar="DIR", help=OUT_HELP)
    cp.add_argument(
        "--labels", nargs="+", help="models and their order (default: all, alphabetical)"
    )
    cp.add_argument("--reference", help="reference model (default: the first)")
    c = sub.add_parser(
        "catalog",
        help="aggregate the results of many inversions: catalog.html over every reported run DIR/<id>/result.json "
        "(a labeled run, under <id>/models/<label>/, is not listed; the input event list is not a catalog)",
    )
    c.add_argument(
        "out", type=Path, metavar="DIR", help="the output directory of the runs (--out of run)"
    )
    rpl = sub.add_parser(
        "replay", help="run a recorded run again, offline, and compare the records"
    )
    rpl.add_argument("result", type=Path, help="the result.json of the run")
    rpl.add_argument(
        "--out", type=Path, metavar="DIR", required=True, help="output directory for the replay"
    )
    rpl.add_argument("--data", type=Path, help="the download, if it has moved since the run")
    rpl.add_argument("--device", help="default: the device the run used")
    rpl.add_argument(
        "--catalogs",
        type=Path,
        metavar="DIR",
        help="directory holding the reference catalog files (default XCAP3D_CATALOGS)",
    )
    db = sub.add_parser(
        "db", help="a Green's-function database: freeze it (release) or check it (check)"
    )
    dbs = db.add_subparsers(dest="dbcmd", required=True)
    w = dbs.add_parser(
        "release",
        help="freeze a database: write release.json with the per-file sha256, the filter families and the provenance "
        "of every station store; inputs: the database directory (and --acceptance records); writes only the "
        "release file, never a store; fails when a store breaks the store contract (as db check) or the stores disagree on "
        "an attribute a release shares (engine, pulse, lattice, Q)",
    )
    w.add_argument("db", type=Path)
    w.add_argument(
        "--out",
        type=Path,
        help="where to write it (default <db>/release.json, the previous one kept as release-v<N>.json)",
    )
    w.add_argument("--name", help="release name (default: the directory name)")
    w.add_argument(
        "--version", type=int, help="default: 1, or the version of <db>/release.json plus 1"
    )
    w.add_argument(
        "--no-hash", action="store_true", help="sizes and fingerprints only, no whole-file sha256"
    )
    w.add_argument("--workers", type=int, default=1, help="files hashed at once")
    w.add_argument(
        "--acceptance", type=Path, help="folder of the stations' rollout_<STA>.json records"
    )
    ck = dbs.add_parser(
        "check",
        help="is the database intact and conforming: every store against the store contract (xcap3d.gf.storeformat) "
        "and the files against release.json (sizes and fingerprints; --full re-hashes every file); inputs: the "
        "database directory and its release file; writes nothing (--json goes outside the database); fails when a "
        "file changed since the release or a store breaks the contract",
    )
    ck.add_argument("db", type=Path)
    ck.add_argument("--release", type=Path, help="default <db>/release.json")
    ck.add_argument(
        "--no-release",
        action="store_true",
        help="the database has no release file: check the store contract only",
    )
    ck.add_argument("--full", action="store_true", help="also the whole-file sha256 (slow)")
    ck.add_argument(
        "--no-data", action="store_true", help="attributes only: do not read one column per force"
    )
    ck.add_argument("--workers", type=int, default=1, help="stores checked at once (processes)")
    ck.add_argument(
        "--json", type=Path, help="write the problems to this file (outside the database)"
    )
    b = sub.add_parser(
        "build",
        help="build one station's Green's-function store with SPECFEM (xcap3d.build): the velocity model, then the "
        "simulations, conversion and acceptance",
    )
    bs = b.add_subparsers(dest="buildcmd", required=True)
    bm = bs.add_parser(
        "model",
        help="the station box's velocity model: UCVM through genmod_parallel.py, the pad pass, the checks -> "
        "model_report.json",
    )
    bg = bs.add_parser(
        "sgt",
        help="mesh, databases, the three forces, conversion and acceptance -> an inversion-ready store; steps "
        "already done are skipped",
    )
    for q in (bm, bg):
        q.add_argument("station")
        q.add_argument(
            "--site",
            type=Path,
            help="the site file: data root, SPECFEM, UCVM (default XCAP3D_SITE; template "
            "xcap3d/build/site.example.toml)",
        )
        q.add_argument(
            "--config",
            type=Path,
            help="station-box configuration (default xcap3d/build/stationbox.json)",
        )
    bm.add_argument(
        "--allow-partial",
        action="store_true",
        help="build a box with < min_usable_frac usable lattice",
    )
    bg.add_argument("--gpus", help="use only these GPUs, e.g. 2,3,7 (default: any free GPU)")
    bg.add_argument("--gpus-per-force", type=int, choices=(1, 2), help="GPUs per force (default 2)")
    bg.add_argument(
        "--keep-raw",
        action="store_true",
        help="keep the raw solver output and databases after conversion",
    )
    bg.add_argument(
        "--convert-only",
        action="store_true",
        help="redo conversion and acceptance from the kept raw solver output; never starts the solver",
    )
    vz = sub.add_parser("viz", help="pictures of a store")
    vzs = vz.add_subparsers(dest="vizcmd", required=True)
    wf = vzs.add_parser(
        "wavefield",
        help="a frame strip (PNG) and an animation (mp4 with imageio-ffmpeg, else GIF) of the strain at the nodes of "
        "a store, one slice, one force, filtered causally; by reciprocity a frame at time t is, at every node, "
        "the sample at t of that node's Green's function",
    )
    wf.add_argument("store", help="a station store (.h5), or a station name with --gf")
    wf.add_argument("--gf", type=Path, help="database directory, to find a station by name")
    wf.add_argument("--force", choices=("fx", "fy", "fz"), default="fz")
    wf.add_argument(
        "--quantity",
        default="vol",
        help="vol (xx + yy + zz) or a component: xx yy zz xy xz yz (default vol)",
    )
    wf.add_argument(
        "--slice",
        default="depth=15",
        help="depth=KM (horizontal), x=KM (section at KM east of the station, along y) or diag (section along "
        "ix = iy) (default depth=15)",
    )
    wf.add_argument("--band", choices=("raw", "pnl", "surface"), default="raw")
    wf.add_argument(
        "--mag", type=float, default=4.5, help="magnitude of the trapezoid and the band corners"
    )
    wf.add_argument("--compare", help="a second store on the same lattice: panels A, B and A - B")
    wf.add_argument("--out", type=Path, default=Path("wavefield"))
    wf.add_argument("--frames", type=int, default=60, help="frames of the animation (default 60)")
    a = p.parse_args(argv)
    if "mt" in a:
        a.mt = MT_SHORT.get(a.mt, a.mt)
    if a.cmd == "download":
        return download_event(a.event_id, a.out)
    if a.cmd == "process":
        return process_event(a.event_id, a.out)
    if a.cmd == "build":
        from .build import config

        over = dict(allow_partial=True) if getattr(a, "allow_partial", False) else {}
        if a.buildcmd == "sgt":
            over.update(keep_raw_after_store=a.keep_raw or None, gpus_per_force=a.gpus_per_force)
            if a.gpus:
                over["gpus_allowed"] = [int(g) for g in a.gpus.split(",")]
        config.set_site(a.site)
        box = config.Box(a.station, config.load_config(a.config or config.CONFIG, **over))
        if a.buildcmd == "model":
            from .build.model import model

            out = model(box)
        else:
            from .build.specfem import sgt

            out = sgt(box, a.convert_only)
        print(json.dumps(out, indent=1, default=str)[:3000])
        return out
    if a.cmd == "viz":
        from .report.wavefield import wavefield

        kind, _, value = a.slice.partition("=")
        if kind not in ("depth", "x", "diag") or (kind == "diag") == bool(value):
            raise SystemExit(f"--slice {a.slice}: use depth=KM, x=KM or diag")
        paths = wavefield(
            a.store,
            a.gf,
            a.force,
            a.quantity,
            (kind, float(value) if value else None),
            a.band,
            a.mag,
            a.compare,
            a.out,
            a.frames,
        )
        print(*paths, sep="\n")
        return paths
    if a.cmd == "replay":
        diff = replay(a.result, a.out, a.data, a.device, a.catalogs)
        print("identical" if not diff else "differs:\n  " + "\n  ".join(diff))
        if diff:
            sys.exit(1)
        return diff
    if a.cmd == "db" and a.dbcmd == "release":
        try:
            path = write_release(
                a.db, a.out, a.name, a.version, not a.no_hash, a.workers, a.acceptance
            )
        except ReleaseError as e:
            raise SystemExit(f"db release: {e}")
        print(path)
        return path
    if a.cmd == "db":
        from .gf.storeformat import check_db

        if a.json and a.json.resolve().is_relative_to(a.db.resolve()):
            raise SystemExit(
                f"--json {a.json}: not inside the database, which the check never writes to"
            )
        if a.no_release and a.release:
            raise SystemExit("--no-release and --release exclude each other")
        rel = None
        if not a.no_release:
            try:
                rel = read_release(a.db, a.release)
            except ReleaseError as e:
                raise SystemExit(f"{e} (--no-release checks the store contract only)")
        res = check_db(a.db, not a.no_data, a.workers)
        for s, b in res["stores"].items():
            if b:
                print(f"{s}: " + "; ".join(b))
        for b in res["database"]:
            print(b)
        failed = [s for s, b in res["stores"].items() if b]
        n = len(res["stores"])
        print(f"{a.db}: {n - len(failed)} of {n} stores meet the store contract")
        changed = res["stamps_before"] != res["stamps_after"]
        if changed:
            print(f"{a.db}: files changed while the check ran; its results do not hold")
        bad = {}
        if rel is not None:
            bad = {s: b for s in rel["stations"] if (b := station_problems(a.db, rel, s, a.full))}
            for s, b in bad.items():
                print(f"{s}: " + "; ".join(b))
            print(
                f"{rel['id']}: {len(rel['stations']) - len(bad)} of {len(rel['stations'])} stations match"
            )
        res.update(release=rel and rel["id"], release_problems=bad, files_changed=changed)
        if a.json:
            a.json.write_text(json.dumps(res, indent=1))
        if failed or res["database"] or bad or changed:
            sys.exit(1)
        return res
    if getattr(a, "label", None) and not re.fullmatch(report.LABEL_RE, a.label):
        raise SystemExit(f"--label {a.label!r}: letters, digits and . _ + - only")
    if a.cmd == "report":
        out = out_dir(a.out) / a.event_id / ("models/" + a.label if a.label else "")
        if not (out / "result.json").exists():
            raise SystemExit(f"{a.event_id}: no result.json in {out}")
        rec = json.loads((out / "result.json").read_text())
        if rec.get("status") in ("no_solution", "error"):
            what = "no solution" if rec["status"] == "no_solution" else "the run failed"
            raise SystemExit(f"{a.event_id}: {what}, so no report ({rec['reason']})")
        try:
            page = report.write_event_report(rec, out)
        except ValueError as e:  # a record without its report part (product_error)
            raise SystemExit(str(e))
        print(page)
        return page
    if a.cmd == "compare":
        page = report.write_compare(out_dir(a.out) / a.event_id, a.labels, a.reference)
        print(page)
        return page
    if a.cmd == "catalog":
        rows, skipped, failed, errors = [], [], [], []
        for f in sorted(a.out.glob("*/result.json")):
            rec = json.loads(f.read_text())
            if rec.get("schema") == record.SCHEMA and (f.parent / "report.html").exists():
                rows.append(report.catalog_row(rec, f.parent, a.out))
            elif rec.get("status") == "no_solution":
                failed.append(f.parent.name)
            elif rec.get("status") == "error":
                errors.append(f.parent.name)
            else:
                skipped.append(f.parent.name)
        page = report.write_catalog(rows, a.out)
        print(
            f"{page}: {len(rows)} events"
            + (f" ({len(skipped)} runs without a report skipped)" if skipped else "")
            + (f"; no solution: {', '.join(failed)}" if failed else "")
            + (f"; error: {', '.join(errors)}" if errors else "")
        )
        return page
    settings = dict(
        jackknife=a.jackknife,
        edge_margin_km=a.edge_margin_km,
        near_field_cells=a.near_field_cells,
        write_report=a.report,
        label=a.label,
        out=a.out,
        device=a.device,
        release=a.release,
        catalogs=a.catalogs,
        mt=a.mt,
        refine=a.refine,
        grid_step=a.grid_step,
        refine_step=a.refine_step,
    )
    if a.cmd == "invert":
        return invert_event(a.event_id, a.gf, **settings)
    return run_event(a.event_id, a.gf, **settings)


def entry():
    """Console script entry point; sets the exit status from the record.

    The generated `sys.exit(main())` would print the returned dict and exit 1, so this returns a status instead: 2 when
    the inversion ran but its record could not be built, 3 when there is no solution.
    """
    result = main()
    status = result.get("status") if isinstance(result, dict) else None
    if status == "product_error":
        sys.exit(2)
    if status == "no_solution":
        sys.exit(3)
