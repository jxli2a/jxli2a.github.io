Source code for xcap3d.report.report

"""Reports: the per-event `report.html`, the `compare.html` of one event over several Green's-function sets, and the
`catalog.html` index, from the records of record.py.

Replaces `genFIGURE.csh` + `genREPORT.csh`. No number is changed: the fit figure re-reads the waveforms at the chosen
depth without inverting. The figures are in `viz.py`.
"""

from __future__ import annotations

import json
import math
from pathlib import Path

import numpy as np

from ..core.constants import COMPONENTS
from ..core.mt import kagan_angle
from . import viz
from .record import BAND_NOTE, SCHEMA, now, station_vrs

LABEL_RE = r"[A-Za-z0-9][A-Za-z0-9._+-]*"


def vr_ranks(stations):
    """{name: rank} of the stations with a VR_s (`vr`), 1 the best."""
    vr = sorted((s for s in stations if "vr" in s), key=lambda s: -s["vr"])
    return {s["name"]: k + 1 for k, s in enumerate(vr)}


def table_order(stations):
    """The station table: by VR_s, best first; used stations without one (a record from before VR_s), then the
    discarded ones, nearest first. Returns (rows, number of used rows)."""
    near = sorted(stations, key=lambda s: s["dist_km"])
    ranked = sorted((s for s in near if "vr" in s), key=lambda s: -s["vr"])
    used = ranked + [s for s in near if "vr" not in s and s["status"] == "used"]
    return used + [s for s in near if s["status"] not in ("used", "not loaded")], len(used)


def fits_rows(rec, path):
    """`fit_rows` for a production run: the windows of the chosen depth from the run's fits.h5, with the misfit's
    trace weights, in the record's order (nearest first), and each station's VR_s rank."""
    from ..core.invert import station_weight
    from .products import read_fits

    fits = read_fits(path, rec["depth"])
    rank = vr_ranks(rec["stations"])
    rows = []
    for s in rec["stations"]:
        f = fits.get(s["name"])
        if f is None:
            continue
        w = station_weight(f["dist"])
        rows.append(
            dict(
                station=s,
                rank=rank.get(s["name"]),
                obs=[np.asarray(o, float) * w[j] for j, o in enumerate(f["obs"])],
                syn=[np.asarray(y, float) * w[j] for j, y in enumerate(f["syn"])],
            )
        )
    return rows


def num(v, fmt="%.1f", none="–"):
    """Jinja filter: printf format (signed formats print their sign once), a true minus sign, no '-0'."""
    if v is None:
        return none
    t = fmt % v
    return (fmt % 0.0 if float(t) == 0 else t).replace("-", "\u2212")


def sci(v):
    e = math.floor(math.log10(v))
    return "%.2f×10<sup>%d</sup>" % (v / 10**e, e)


def _env():
    import jinja2

    env = jinja2.Environment(
        loader=jinja2.FileSystemLoader(Path(__file__).parent / "templates"),
        autoescape=jinja2.select_autoescape(["html"]),
        trim_blocks=True,
        lstrip_blocks=True,
    )
    env.filters["num"], env.filters["sci"] = num, sci
    env.globals["band_note"] = BAND_NOTE
    return env


def depth_rows(rec):
    """The per-depth table under the depth figure, transposed (a column per depth): (label, cells)."""
    scan = rec["depth_scan"]
    s0 = min(r["score"] for r in scan)
    rows = [
        ("Depth (km)", [num(r["depth"]) for r in scan]),
        ("Misfit above best (%)", [num(r["rel_misfit"], "%.0f") for r in scan]),
        ("Mean CC", [num(r["ave_cc"], "%.2f") for r in scan]),
        ("Score above minimum", [num(r["score"] - s0, "%.1f") for r in scan]),
        ("Weight sum", [num(r["ncmp_used"], "%.0f") for r in scan]),
    ]
    rows += [(k.capitalize(), [num(r[k], "%.0f") for r in scan]) for k in ("strike", "dip", "rake")]
    return rows


def page_context(rec, figs, has_map, pfx="", base=""):
    """The variables `event_body.html` renders for one record.

    `figs` maps beachball, depth, fits, map, ours and ref_<key> to image sources (file names or data URIs) and may carry
    `fits_note`; `pfx` prefixes the section ids, and `base` the record's own files (`models/<label>/` in the comparison
    page).
    """
    s = rec["solution"]
    sdr = (s["strike"], s["dip"], s["rake"])
    nd = rec.get("non_dc")
    counts = {
        k: sum(x["status"] == k for x in rec["stations"])
        for k in ("used", "discarded", "not loaded", "downloaded")
    }
    table, n_used = table_order(rec["stations"])
    return dict(
        rec=rec,
        ev=rec["event"],
        s=s,
        dc=rec["depth_confidence"],
        vd=rec["verdict"],
        scan=rec["depth_scan"],
        prov=rec["provenance"],
        refs=rec["references"],
        missing=rec.get("references_missing", []),
        figs=figs,
        depth_rows=depth_rows(rec),
        aux=" / ".join(num(v, "%.0f") for v in aux_plane(sdr)),
        counts=counts,
        table=table,
        n_used=n_used,
        comps=COMPONENTS,
        settings=rec["settings"],
        qc=rec["qc"],
        has_map=has_map,
        pfx=pfx,
        base=base,
        unc=None if "skipped" in (rec.get("uncertainty") or {}) else rec.get("uncertainty"),
        jk_skipped=(rec.get("uncertainty") or {}).get(
            "skipped"
        ),  # a jackknife that could not run, and why
        nd=nd,
        nd_kagan=kagan_angle((nd["strike"], nd["dip"], nd["rake"]), sdr) if nd else None,
    )


FIGS = dict(
    beachball="beachball.png",
    depth="depth.png",
    fits="fits.png",
    map="map.png",
    ours="bb_ours.png",
    lune="lune.png",
    nondc="bb_nondc.png",  # the non-DC moment tensor, and its double couple
    nondc_dc="bb_nondc_dc.png",
)


def _figs(out, rec):
    """Figure files present in `out` for `rec` (None where absent), and the reason a missing fit figure is missing."""
    figs = {k: (v if (out / v).exists() else None) for k, v in FIGS.items()}
    figs.update(
        {
            f"ref_{r['key']}": f"bb_{r['key']}.png"
            for r in rec["references"]
            if (out / f"bb_{r['key']}.png").exists()
        }
    )
    if (out / "fits_error.txt").exists() and not figs["fits"]:
        figs["fits_note"] = (out / "fits_error.txt").read_text().strip()
    return figs


def inline_figs(out, rec):
    """`_figs` with each figure as a data URI: the page shows them without the PNG files beside it."""
    return {
        k: (data_uri(out / v) if k != "fits_note" and v else v) for k, v in _figs(out, rec).items()
    }

def write_event_report(result, out_dir, event_meta=None, refs=None, figures=True):
    """Write `report.html` and its figures from a result record.

    The page carries its figures as data URIs, so it shows them anywhere on its own; the PNG files stay for the
    catalog page and for reuse.

    A labeled model run (`greens` key, `<id>/models/<label>/`) links back to the event's comparison page.

    Args:
        result: the result.json record; must be of schema SCHEMA.
        out_dir: directory for the page and figures.
        event_meta: overrides the record's `event`.
        refs: overrides the record's `references`.
        figures: draw the figures; False re-renders the page around existing PNGs.

    Returns:
        Path of report.html.

    Raises:
        ValueError: the record is not of schema SCHEMA.
    """
    if result.get("schema") != SCHEMA:
        raise ValueError(
            f"{result.get('event_id')}: result.json has no report record (schema {SCHEMA})"
            + (
                f": {result['report_error']}"
                if result.get("report_error")
                else "; re-run with --report"
            )
        )
    out = Path(out_dir)
    out.mkdir(parents=True, exist_ok=True)
    rec = dict(
        result,
        event=event_meta or result["event"],
        references=result["references"] if refs is None else refs,
    )
    if (out / "fits.h5").exists() and not any("vr" in st for st in rec["stations"]):
        # a record from before VR_s: the stations' VR_s from its fits.h5, for this page only
        from .products import read_fits

        rec["stations"] = [dict(st) for st in rec["stations"]]
        station_vrs(
            rec["stations"], read_fits(out / "fits.h5", rec["depth"]), rec["settings"]["dt_s"]
        )
    s = rec["solution"]
    sdr = (s["strike"], s["dip"], s["rake"])
    has_map = viz.map_available()
    if figures:
        viz.fig_beachball(sdr, out / FIGS["beachball"])
        viz.fig_beachball(sdr, out / FIGS["ours"], size_in=0.42)
        viz.fig_depth(rec, out / FIGS["depth"])
        nd = rec.get("non_dc")
        if nd:
            viz.fig_lune(nd, out / FIGS["lune"])
            viz.fig_beachball(nd["mt_ned"], out / FIGS["nondc"], size_in=1.2)
            viz.fig_beachball(
                (nd["strike"], nd["dip"], nd["rake"]), out / FIGS["nondc_dc"], size_in=1.2
            )
        else:
            for k in ("lune", "nondc", "nondc_dc"):
                (out / FIGS[k]).unlink(missing_ok=True)
        (out / "fits_error.txt").unlink(missing_ok=True)
        if (out / "fits.h5").exists():
            try:
                viz.fig_fits(fits_rows(rec, out / "fits.h5"), out / FIGS["fits"])
            except Exception as e:  # e.g. no GFs at the chosen depth under this model's label
                (out / FIGS["fits"]).unlink(missing_ok=True)
                (out / "fits_error.txt").write_text(
                    f"waveforms could not be re-read ({type(e).__name__}: {e})\n"
                )
        if has_map:
            viz.fig_map(rec, out / FIGS["map"])
        else:
            (out / FIGS["map"]).unlink(missing_ok=True)
        for r in rec["references"]:
            if r["strike"] is not None:
                viz.fig_beachball(
                    (r["strike"], r["dip"], r["rake"]),
                    out / f"bb_{r['key']}.png",
                    color=viz.REF,
                    size_in=0.42,
                )
    label = (rec.get("greens") or {}).get("label")
    links = (
        [("catalog", "../../../catalog.html"), ("all models", "../../compare.html")]
        if label
        else [("catalog", "../catalog.html")]
    )
    html = (
        _env()
        .get_template("event.html")
        .render(**page_context(rec, inline_figs(out, rec), has_map), links=links, label=label)
    )
    (out / "report.html").write_text(html)
    return out / "report.html"




def aux_plane(sdr):
    from obspy.imaging.beachball import aux_plane as ap

    return ap(*sdr)


def catalog_row(rec, report_dir, root):
    """One line of the catalog for the record in `report_dir`; paths relative to the catalog's `root`."""
    s, ev = rec["solution"], rec["event"]
    rel = Path(report_dir).resolve().relative_to(Path(root).resolve())
    gji = next((r for r in rec["references"] if r["key"] == "gji"), None)
    return dict(
        event_id=rec["event_id"],
        time=ev["time"][:19],
        place=ev.get("place", ""),
        lat=ev["lat"],
        lon=ev["lon"],
        depth=s["depth_km"],
        mw=s["mw"],
        strike=s["strike"],
        dip=s["dip"],
        rake=s["rake"],
        vr=s["vr"],
        n_stations=s["n_stations"],
        n_available=s["n_available"],
        caveats=rec["verdict"]["caveats"],
        band=rec["depth_confidence"]["band_km"],
        **jackknife_cols(rec.get("uncertainty")),
        kagan_gji=gji["kagan"] if gji else None,
        report=(rel / "report.html").as_posix(),
        thumb=(rel / "beachball.png").as_posix(),
    )


def jackknife_cols(u):
    """Catalog columns of a record's `uncertainty` (None where the jackknife did not run or the depth part was off)."""
    u = u or {}
    return dict(
        jk_depth_se=u.get("depth_se_km"),
        jk_p50=u.get("mechanism_jackknife_p50"),
        jk_p90=u.get("mechanism_jackknife_p90"),
    )


def cat_stats(rows):
    mw = [r["mw"] for r in rows]
    kg = [r["kagan_gji"] for r in rows if r["kagan_gji"] is not None]
    tally = {}
    for r in rows:
        for c in r["caveats"] or ["well constrained"]:
            tally[c] = tally.get(c, 0) + 1
    return dict(
        n=len(rows),
        mw=(min(mw), max(mw)) if mw else None,
        median_vr=float(np.median([r["vr"] for r in rows])) if rows else None,
        years=(rows[0]["time"][:4], rows[-1]["time"][:4]) if rows else None,
        median_kagan_gji=float(np.median(kg)) if kg else None,
        n_gji=len(kg),
        n_gji_over20=sum(k > 20 for k in kg),
        n_clean=sum(not r["caveats"] for r in rows),
        tally=sorted(tally.items(), key=lambda kv: -kv[1]),
    )


def render_catalog(views, out):
    """`views`: [dict(rows, map)]."""
    has_map = any(v["map"] for v in views)
    for v in views:
        v["rows"] = sorted(v["rows"], key=lambda r: r["time"])
        v["stats"] = cat_stats(v["rows"])
        if v["map"] and v["rows"]:
            viz.fig_catalog_map(v["rows"], out / v["map"])
        elif v["map"]:
            v["map"] = None
    html = (
        _env()
        .get_template("catalog.html")
        .render(
            views=views,
            has_map=has_map,
            generated=now(),
        )
    )
    (out / "catalog.html").write_text(html)
    return out / "catalog.html"


def write_catalog(index_rows, out_dir):
    """Write `catalog.html` (and `catalog_map.png` when the map data is cached) over catalog rows.

    Args:
        index_rows: rows from `catalog_row`.
        out_dir: output directory.

    Returns:
        Path of catalog.html.
    """
    out = Path(out_dir)
    out.mkdir(parents=True, exist_ok=True)
    m = "catalog_map.png" if index_rows and viz.map_available() else None
    return render_catalog([dict(rows=list(index_rows), map=m)], out)


def model_labels(event_dir, labels=None):
    """Labels with a result under `event_dir/models`, runs without a solution or that failed left out.

    In the order given (unknown ones dropped), else alphabetical.
    """
    found = sorted(
        p.parent.name
        for p in (Path(event_dir) / "models").glob("*/result.json")
        if json.loads(p.read_text()).get("status") not in ("no_solution", "error")
    )
    return [label for label in labels if label in found] if labels else found


def report_current(model_dir):
    """True when `model_dir/report.html` exists and is not older than its `result.json`."""
    d = Path(model_dir)
    return (d / "report.html").exists() and (d / "report.html").stat().st_mtime >= (
        d / "result.json"
    ).stat().st_mtime


def clear_report(model_dir):
    """Remove a model's report.html, figures, fits.h5 and fits note, which belong to the previous result.json.

    Only the files the report writes: an event directory may hold other figures.
    """
    d = Path(model_dir)
    for f in [
        d / "report.html",
        d / "fits_error.txt",
        d / "fits.h5",
        *map(d.joinpath, FIGS.values()),
        *d.glob("bb_*.png"),
    ]:
        f.unlink(missing_ok=True)


def summary(rec):
    """Strip values of one model's record; flat (failed-report) records work too."""
    s = rec.get("solution")
    return dict(
        sdr=(rec["strike"], rec["dip"], rec["rake"]),
        mw=rec["mw"],
        depth=rec["depth"],
        vr=rec["vr"],
        band=rec["depth_confidence"]["band_km"] if s else None,
        caveats=rec["verdict"]["caveats"] if s else None,
        **jackknife_cols(rec.get("uncertainty")),
        n_stations=s["n_stations"] if s else None,
        error=rec.get("report_error"),
    )


def delta(ref, other):
    """other minus ref: Kagan angle, Mw, depth, VR."""
    a, b = summary(ref), summary(other)
    return dict(
        kagan=kagan_angle(a["sdr"], b["sdr"]),
        dmw=b["mw"] - a["mw"],
        ddepth=b["depth"] - a["depth"],
        dvr=b["vr"] - a["vr"],
    )


def data_uri(path):
    import base64

    return "data:image/png;base64," + base64.b64encode(Path(path).read_bytes()).decode()

def write_compare(event_dir, labels=None, reference=None):
    """Write `<event_dir>/compare.html`: every model of one event in one self-contained page.

    A strip lists all models (mechanism, Mw, depth, VR, and Kagan / dMw / ddepth to the reference) above the event page
    of the selected model, switched by the drop-down, the arrow keys or the strip; `#m=<label>&ref=<label>` in the URL
    selects. Figures are embedded as data URIs; records without figures get them first (`write_event_report`).

    Args:
        event_dir: the event directory holding `models/<label>/result.json`.
        labels: models to include, in this order; default all, alphabetical.
        reference: the reference model; default the first label.

    Returns:
        Path of compare.html.

    Raises:
        ValueError: the event has no model result.
    """
    event_dir = Path(event_dir)
    labels = model_labels(event_dir, labels)
    if not labels:
        raise ValueError(f"{event_dir}: no models/<label>/result.json")
    reference = reference if reference in labels else labels[0]
    recs = {
        label: json.loads((event_dir / "models" / label / "result.json").read_text())
        for label in labels
    }
    has_map = viz.map_available()
    env, models = _env(), []
    for label in labels:
        rec, d = recs[label], event_dir / "models" / label
        if rec.get("schema") == SCHEMA:
            if not report_current(d):  # missing, or older than its result.json
                write_event_report(rec, d)
            figs = inline_figs(d, rec)
            body = env.get_template("event_body.html").render(
                **page_context(rec, figs, has_map, pfx=f"{label}-", base=f"models/{label}/")
            )
            thumb = figs["beachball"]
        else:
            body = env.get_template("model_missing.html").render(rec=rec, label=label)
            thumb = None
        models.append(
            dict(
                label=label,
                rec=rec,
                sum=summary(rec),
                body=body,
                thumb=thumb,
                greens=rec.get("greens") or {},
                deltas={r: delta(recs[r], rec) for r in labels},
            )
        )
    ev = next((m["rec"]["event"] for m in models if m["rec"].get("schema") == SCHEMA), None)
    html = env.get_template("compare.html").render(
        models=models,
        labels=labels,
        reference=reference,
        ev=ev,
        event_id=recs[labels[0]]["event_id"],
        generated=now(),
    )
    (event_dir / "compare.html").write_text(html)
    return event_dir / "compare.html"
