Source code for xcap3d.report.products

"""`fits.h5`: the observed and synthetic windows of a production run at every scanned depth.

Written from the stations and solutions the run held, so a report or viewer can draw the fits without re-deriving
them. Layout: file attributes format_version, dt and components; one group per depth (`"11.5"`; attributes depth,
strike, dip, rake, m0, mw); one subgroup per loaded station (attributes dist, weight, shift, cc and t0, the window
start [s after the origin], one value per component, and in_search); and in it one dataset per component (Pz, Pr, Sz,
Sr, Sh) of shape (2, n): the observed window and the unshifted synthetic of that depth's solution, in the data's
units (m0 times the Green's functions times the tensor, as `invert.grid_search` defines it). The shift is the lag at
which data sample t + shift / dt best matches synthetic sample t. Stations the amplitude screen or the pre-pass left
out have in_search false, weight 0 and NaN shift and CC. `fits_digest` is the sha256 of the content; result.json
names it, so a record and its fits.h5 can be matched.
"""

import hashlib
import os
from pathlib import Path

import h5py
import numpy as np

from ..core.constants import COMPONENTS, MAX_SHIFT_S

FORMAT_VERSION = 1  # of the layout above


def fits_of(stations, sol, removed=(), meta=None):
    """The fits of one scanned depth: its solution and, per station, the windows and per-component numbers.

    `removed` are stations loaded at this depth but left out of the search; `meta` is the download's, for the window
    starts.
    """
    rows = {}
    nan = np.full(5, np.nan)
    for i, st in enumerate([*stations, *removed]):
        searched = i < len(stations)
        rows[st.name] = dict(
            dist=float(st.dist),
            weight=np.asarray(st.on_off, dtype=float).copy() if searched else np.zeros(5),
            shift=np.asarray(sol.per_comp_shift[i], dtype=float) if searched else nan,
            cc=np.asarray(sol.per_comp_cc[i], dtype=float) if searched else nan,
            t0=np.array([meta[st.name][k] for k in ("t1", "t1", "t3", "t3", "t3")], dtype=float)
            if meta
            else nan,
            in_search=searched,
            obs=[np.asarray(o, dtype=np.float32) for o in st.obs],
            syn=[(np.asarray(g) @ sol.mt * sol.m0).astype(np.float32) for g in st.grn],
        )
    head = dict(strike=sol.strike, dip=sol.dip, rake=sol.rake, m0=sol.m0, mw=sol.mw)
    return dict(head={k: float(v) for k, v in head.items()}, stations=rows)


NUMBERS = ("dist", "weight", "shift", "cc", "t0")

def fits_digest(fits):
    """sha256 of the content of fits, the same for the run's arrays and for the file written from them.

    Args:
        fits: {depth: fits_of(...)}, or `read_all` of a file.

    Returns:
        Hex digest.
    """
    h = hashlib.sha256()
    for depth in sorted(fits):
        d = fits[depth]
        h.update(
            np.asarray([depth, *(d["head"][k] for k in sorted(d["head"]))], dtype=float).tobytes()
        )
        for name in sorted(d["stations"]):
            r = d["stations"][name]
            h.update(name.encode() + bytes([bool(r["in_search"])]))
            for k in NUMBERS:
                h.update(np.asarray(r[k], dtype=float).tobytes())
            for a in [*r["obs"], *r["syn"]]:
                h.update(np.asarray(a, dtype=np.float32).tobytes())
    return h.hexdigest()

def write_fits(path, fits, dt):
    """Write the fits of a run to `path`, through a temporary file.

    Args:
        path: the fits.h5 file.
        fits: {depth: fits_of(...)}.
        dt: sample interval [s].
    """
    path = Path(path)
    tmp = path.with_name(path.name + ".tmp")
    try:
        with h5py.File(tmp, "w") as f:
            f.attrs.update(format_version=FORMAT_VERSION, dt=dt, components=list(COMPONENTS))
            for depth, d in fits.items():
                g = f.create_group(f"{depth:g}")
                g.attrs.update(depth=depth, **d["head"])
                for name, r in d["stations"].items():
                    s = g.create_group(name)
                    s.attrs.update({k: r[k] for k in NUMBERS}, in_search=r["in_search"])
                    for c, o, y in zip(COMPONENTS, r["obs"], r["syn"]):
                        s.create_dataset(c, data=np.stack([o, y]), compression="gzip")
        os.replace(tmp, path)
    finally:
        tmp.unlink(missing_ok=True)




def station_fits(s):
    pairs = [s[c][()] for c in COMPONENTS]
    return dict(
        {k: s.attrs[k] for k in NUMBERS},
        in_search=bool(s.attrs["in_search"]),
        obs=[p[0] for p in pairs],
        syn=[p[1] for p in pairs],
    )


def read_fits(path, depth):
    """{station: dict(dist, weight, shift, cc, t0, in_search, obs, syn)} at one depth of a `fits.h5`."""
    with h5py.File(path, "r") as f:
        return {name: station_fits(s) for name, s in f[f"{float(depth):g}"].items()}


def read_all(path):
    """Every depth of a `fits.h5`, in the form `fits_of` gives."""
    with h5py.File(path, "r") as f:
        return {
            float(g.attrs["depth"]): dict(
                head={k: float(g.attrs[k]) for k in ("strike", "dip", "rake", "m0", "mw")},
                stations={name: station_fits(s) for name, s in g.items()},
            )
            for g in f.values()
        }


def correlation(obs, syn, lag):
    """Normalized correlation of data sample t + lag with synthetic sample t, as the search measures it.

    None when the data or the synthetic has no energy: 0/0 would be NaN, which strict JSON cannot hold.
    """
    obs, syn = np.asarray(obs, dtype=float), np.asarray(syn, dtype=float)
    energy = (obs @ obs) * (syn @ syn)
    if energy == 0:
        return None
    n = len(obs)
    a, b = (obs[lag:], syn[: n - lag]) if lag >= 0 else (obs[: n + lag], syn[-lag:])
    return float(a @ b / np.sqrt(energy))


def station_vr(f, dt):
    """VR_s of one station's fit: 1 - sum err_c / sum |d_c|^2 over its used components (weight > 0); None if none.

    |d_c|^2 is the distance-scaled data energy of the misfit (rec2 of invert.precompute) and err_c the component's misfit
    at the solution, |d_c|^2 + |s_c|^2 - 2 (d_c * s_c)(shift), invert.grid_search's per_comp_err, both recomputed from
    the stored windows (float32, so to about 1e-6). `f`: one station of `fits_of` or `read_fits`.
    """
    from ..core.invert import station_weight

    w2 = station_weight(f["dist"]) ** 2
    err = rec2 = 0.0
    for j in np.flatnonzero(np.asarray(f["weight"]) > 0):
        d, s = np.asarray(f["obs"][j], dtype=float), np.asarray(f["syn"][j], dtype=float)
        lag, n = int(round(f["shift"][j] / dt)), len(d)
        a, b = (d[lag:], s[: n - lag]) if lag >= 0 else (d[: n + lag], s[-lag:])
        err += w2[j] * (d @ d + s @ s - 2 * (a @ b))
        rec2 += w2[j] * (d @ d)
    return float(1 - err / rec2) if rec2 > 0 else None


def shift_check(f, bands, dt):
    """Per component of one station's fit: the shift bound and whether the shift reached it or a cycle skip is likely.

    The bound is half the shortest period of the component's band. Reports whether the shift sits at the bound or
    beyond half a period, and the best CC within half a period of zero lag (a cycle skip shows as a CC there close to
    the best). None where a component has no weight. Diagnostics only.
    """
    out = dict(
        bound=[], half_period=[], at_bound=[], beyond_half_period=[], cc_within_half_period=[]
    )
    for j in range(5):
        bound, half = float(MAX_SHIFT_S[j]), 0.5 / (bands[1] if j < 2 else bands[3])
        shift = float(f["shift"][j])
        live = f["weight"][j] > 0
        k = int(np.floor(half / dt + 1e-9))
        ccs = [correlation(f["obs"][j], f["syn"][j], lag) for lag in range(-k, k + 1)]
        cc = None if None in ccs else max(ccs)
        out["bound"].append(bound)
        out["half_period"].append(half)
        out["at_bound"].append(bool(abs(shift) >= bound - dt / 2) if live else None)
        out["beyond_half_period"].append(bool(abs(shift) > half) if live else None)
        out["cc_within_half_period"].append(cc if live else None)
    return out
